-- Close out stale in-progress trips
-- 1. Active trips with driver
SELECT t.id, d.name AS driver, t.status, t.distance_km, t.started_at
FROM TRIP t
JOIN DRIVER d ON d.id = t.driver_id
WHERE t.status = 'in_progress'
ORDER BY t.started_at DESC;

-- 2. Close a trip
UPDATE TRIP SET status = 'completed', ended_at = NOW()
WHERE id = 88213;

-- 3. Trips by status
SELECT status, COUNT(*) AS trips
FROM TRIP
GROUP BY status
ORDER BY trips DESC;
